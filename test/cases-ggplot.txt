# ggplot2 against base R (D45).
#
# ggplot2 is not installed in the system R, so a plot cannot be compared with itself.
# What can be compared is what a plot is made of: the counts geom_bar() computes, where
# a stacked or dodged bar starts and ends, which bar comes first, what an axis shows.
# layer_data() gives that as a table, as it does in ggplot2; the base-R side computes
# the same numbers by hand.
#
# Separator is |||   Lines starting with # are comments.
SETUP: d <- data.frame(miasto = c("Kraków","Warszawa","Kraków","Gdańsk","Warszawa","Kraków","Gdańsk","Warszawa","Kraków","Gdańsk"), plec = c("K","M","K","K","M","M","K","M","K","M"), wiek = c(23,34,45,29,51,38,27,42,19,60), ocena = c(4,5,3,4,2,5,4,NA,3,5), stringsAsFactors = FALSE)

# --- geom_bar() counts the rows itself ---
layer_data(ggplot(d, aes(x = miasto)) + geom_bar())$count ||| as.numeric(table(d$miasto))
layer_data(ggplot(d, aes(x = miasto)) + geom_bar())$ymax ||| as.numeric(table(d$miasto))
layer_data(ggplot(d, aes(x = miasto)) + geom_bar())$x ||| as.numeric(seq_along(unique(d$miasto)))
layer_data(ggplot(d, aes(x = wiek > 30)) + geom_bar())$count ||| as.numeric(table(d$wiek > 30))
layer_data(ggplot(d, aes(x = factor(ocena))) + geom_bar())$count ||| as.numeric(table(d$ocena, useNA = "ifany"))
layer_data(ggplot(d, aes(x = ocena)) + geom_bar())$x ||| sort(unique(d$ocena))
layer_data(ggplot(d, aes(x = ocena)) + geom_bar())$count ||| as.numeric(table(d$ocena))

# --- geom_col() takes the numbers it is given ---
layer_data(ggplot(count(d, miasto), aes(miasto, n)) + geom_col())$ymax ||| as.numeric(table(d$miasto))
max(layer_data(ggplot(d, aes(miasto, ocena)) + geom_col())$ymax) ||| max(tapply(d$ocena, d$miasto, sum, na.rm = TRUE))
nrow(layer_data(ggplot(d, aes(miasto, ocena)) + geom_col())) ||| sum(!is.na(d$ocena))
layer_data(ggplot(count(d, miasto), aes(miasto, n)) + geom_col() + geom_text(aes(label = n)), 2)$label ||| as.character(as.numeric(table(d$miasto)))
layer_data(ggplot(count(d, miasto), aes(miasto, n)) + geom_col() + geom_text(aes(label = n)), 2)$y ||| as.numeric(table(d$miasto))

# --- position: stack (the first level on top), fill, dodge ---
layer_data(ggplot(d, aes(x = miasto, fill = plec)) + geom_bar())$ymax ||| tb <- table(d$miasto, d$plec); r <- c(); for (m in rownames(tb)) for (p in colnames(tb)) if (tb[m, p] > 0) r <- c(r, if (p == "K") sum(tb[m, ]) else tb[m, "M"]); as.numeric(r)
layer_data(ggplot(d, aes(x = miasto, fill = plec)) + geom_bar())$ymin ||| tb <- table(d$miasto, d$plec); r <- c(); for (m in rownames(tb)) for (p in colnames(tb)) if (tb[m, p] > 0) r <- c(r, if (p == "K") tb[m, "M"] else 0); as.numeric(r)
layer_data(ggplot(d, aes(x = miasto, fill = plec)) + geom_bar(position = "fill"))$ymax ||| tb <- prop.table(table(d$miasto, d$plec), 1); r <- c(); for (m in rownames(tb)) for (p in colnames(tb)) if (tb[m, p] > 0) r <- c(r, if (p == "K") 1 else tb[m, "M"]); as.numeric(r)
layer_data(ggplot(d, aes(x = miasto, fill = plec)) + geom_bar(position = "dodge"))$ymax ||| tb <- table(d$miasto, d$plec); r <- c(); for (m in rownames(tb)) for (p in colnames(tb)) if (tb[m, p] > 0) r <- c(r, tb[m, p]); as.numeric(r)
layer_data(ggplot(d, aes(x = miasto, fill = plec)) + geom_bar(position = "dodge"))$xmin ||| tb <- table(d$miasto, d$plec); r <- c(); i <- 0; for (m in rownames(tb)) { i <- i + 1; k <- sum(tb[m, ] > 0); j <- 0; for (p in colnames(tb)) if (tb[m, p] > 0) { r <- c(r, i - 0.45 + j * 0.9 / k); j <- j + 1 } }; r
layer_data(ggplot(d, aes(x = miasto, fill = plec)) + geom_bar(position = "dodge"))$xmax ||| tb <- table(d$miasto, d$plec); r <- c(); i <- 0; for (m in rownames(tb)) { i <- i + 1; k <- sum(tb[m, ] > 0); j <- 0; for (p in colnames(tb)) if (tb[m, p] > 0) { j <- j + 1; r <- c(r, i - 0.45 + j * 0.9 / k) } }; r

# --- the order of bars is the order of levels ---
layer_data(ggplot(d, aes(x = factor(miasto, levels = c("Warszawa", "Kraków", "Gdańsk")))) + geom_bar())$count ||| as.numeric(table(factor(d$miasto, levels = c("Warszawa", "Kraków", "Gdańsk"))))
layer_data(ggplot(d, aes(x = fct_infreq(miasto))) + geom_bar())$count ||| as.numeric(sort(table(d$miasto), decreasing = TRUE))
levels(fct_infreq(d$miasto)) ||| names(sort(table(d$miasto), decreasing = TRUE))
levels(reorder(d$miasto, d$wiek)) ||| levels(reorder(d$miasto, d$wiek))
levels(reorder(d$miasto, -d$wiek)) ||| levels(reorder(d$miasto, -d$wiek))
levels(reorder(d$miasto, d$ocena)) ||| levels(reorder(d$miasto, d$ocena))
levels(fct_reorder(d$miasto, d$wiek)) ||| names(sort(tapply(d$wiek, d$miasto, median)))
levels(fct_reorder(d$miasto, d$wiek, .desc = TRUE)) ||| names(sort(tapply(d$wiek, d$miasto, median), decreasing = TRUE))
levels(fct_reorder(d$miasto, d$ocena)) ||| names(sort(tapply(d$ocena, d$miasto, median, na.rm = TRUE)))
levels(fct_rev(factor(d$miasto))) ||| rev(levels(factor(d$miasto)))
levels(fct_relevel(d$miasto, "Warszawa")) ||| c("Warszawa", "Gdańsk", "Kraków")
levels(fct_inorder(d$miasto)) ||| unique(d$miasto)
as.integer(fct_infreq(d$miasto)) ||| as.integer(factor(d$miasto, levels = names(sort(table(d$miasto), decreasing = TRUE))))

# --- geom_bar(stat = "identity") is geom_col() (D62) ---
layer_data(ggplot(count(d, miasto), aes(miasto, n)) + geom_bar(stat = "identity"))$ymax ||| as.numeric(table(d$miasto))
sum(layer_data(ggplot(count(d, miasto, plec), aes(miasto, n, fill = plec)) + geom_bar(stat = "identity"))$ymax) - sum(layer_data(ggplot(count(d, miasto, plec), aes(miasto, n, fill = plec)) + geom_bar(stat = "identity"))$ymin) ||| nrow(d)

# --- geom_bar() on a number axis drops NA rows, as ggplot2 does with a warning (D63) ---
sum(layer_data(ggplot(d, aes(ocena)) + geom_bar())$count) ||| sum(!is.na(d$ocena))
layer_data(ggplot(d, aes(ocena)) + geom_bar())$count ||| as.integer(table(d$ocena))

# --- other kinds of plot (D81): the bins of a histogram, the box of a boxplot, points ---
layer_data(ggplot(d, aes(x = wiek)) + geom_histogram(binwidth = 5))$count ||| as.numeric(table(cut(d$wiek, breaks = seq(17.5, 62.5, by = 5), right = TRUE, include.lowest = TRUE)))
layer_data(ggplot(d, aes(x = wiek)) + geom_histogram(binwidth = 5))$xmin ||| seq(17.5, 57.5, by = 5)
layer_data(ggplot(d, aes(x = wiek)) + geom_histogram(binwidth = 10))$count ||| as.numeric(table(cut(d$wiek, breaks = seq(15, 65, by = 10), right = TRUE, include.lowest = TRUE)))

layer_data(ggplot(d, aes(x = plec, y = wiek)) + geom_boxplot())$ymin ||| as.numeric(tapply(d$wiek, d$plec, function(v) { q <- quantile(v, c(0.25, 0.75)); min(v[v >= q[1] - 1.5 * diff(q)]) }))

# --- facet_wrap() (D82): each panel counts only its own rows, panels in level order ---
layer_data(ggplot(d, aes(x = miasto)) + geom_bar() + facet_wrap(~ plec))$count ||| as.numeric(c(table(d$miasto[d$plec == "K"]), table(d$miasto[d$plec == "M"])))
layer_data(ggplot(d, aes(x = miasto)) + geom_bar() + facet_wrap(~ plec))$PANEL ||| rep(1:2, c(length(table(d$miasto[d$plec == "K"])), length(table(d$miasto[d$plec == "M"]))))
layer_data(ggplot(d, aes(x = plec, y = wiek)) + geom_boxplot() + facet_wrap(~ miasto))$middle ||| unlist(lapply(split(d, d$miasto), function(s) tapply(s$wiek, s$plec, median)), use.names = FALSE)
layer_data(ggplot(d, aes(x = wiek)) + geom_histogram(binwidth = 10) + facet_wrap(~ plec))$count ||| as.numeric(c(table(cut(d$wiek[d$plec == "K"], seq(15, 65, 10), include.lowest = TRUE)), table(cut(d$wiek[d$plec == "M"], seq(15, 65, 10), include.lowest = TRUE))))
